Implementasi Praktis

Membangun Model CNN dengan Python

Contoh implementasi arsitektur CNN sederhana menggunakan TensorFlow / Keras untuk klasifikasi citra.

cnn_model.py
import tensorflow as tf
from tensorflow.keras import layers, models

# 1. Inisialisasi Model Sequential
model = models.Sequential([
    # Layer Konvolusi Pertama + ReLU + MaxPooling
    layers.Conv2D(32, (3, 3), activation='relu', input_shape=(64, 64, 3)),
    layers.MaxPooling2D((2, 2)),

    # Layer Konvolusi Kedua + ReLU + MaxPooling
    layers.Conv2D(64, (3, 3), activation='relu'),
    layers.MaxPooling2D((2, 2)),

    # Flattening & Fully Connected Layer
    layers.Flatten(),
    layers.Dense(64, activation='relu'),
    layers.Dense(10, activation='softmax') # 10 kelas output
])

# 2. Kompilasi Model
model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

model.summary()
